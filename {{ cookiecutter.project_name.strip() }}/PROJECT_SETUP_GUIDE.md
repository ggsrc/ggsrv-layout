# Project Setup Guide

This guide helps you (or an AI assistant) initialize a new project from this template step-by-step.

## 🎯 Goal

By following this guide, you will have a fully functional service with:
- ✅ Database schema and queries
- ✅ gRPC API endpoints
- ✅ Service layer with tests
- ✅ All code generated and compiling
- ✅ Tests passing

## 📋 Prerequisites

Before starting, ensure:
- [ ] Docker is running (for testcontainers or local dependencies)
- [ ] Go 1.27+ is installed
- [ ] Project generated from template: `cookiecutter ggsrv-layout/`

-- name: AddProject :one
-- -- invalidate: [GetAllProjects]
-- -- timeout: 1s
INSERT INTO public.projects (
    name,
    description,
    status,
    created_at,
    updated_at
) VALUES (
    @name,
    @description,
    @status,
    @created_at,
    @updated_at
) RETURNING *;

-- name: GetAllProjects :many
-- -- cache: 300s
-- -- timeout: 500ms
SELECT * FROM public.projects WHERE status = 'ACTIVE';

-- name: GetProjectById :one
-- -- cache: 300s
-- -- timeout: 500ms
SELECT * FROM public.projects WHERE id = @id;

-- name: UpdateProject :one
-- -- invalidate: [GetAllProjects, GetProjectById]
-- -- timeout: 1s
UPDATE public.projects
SET
    name = @name,
    description = @description,
    updated_at = @updated_at
WHERE id = @id
RETURNING *;

-- name: DeleteProject :exec
-- -- invalidate: [GetAllProjects, GetProjectById]
-- -- timeout: 1s
UPDATE public.projects
SET status = 'DELETED', updated_at = @updated_at
WHERE id = @id;

-- name: RefreshID :exec
-- -- timeout: 1s
SELECT setval(pg_get_serial_sequence('projects', 'id'), (SELECT MAX(id) FROM projects));
```{% endraw %}

package svc

import (
    "context"
    "time"

    "google.golang.org/grpc/codes"
    "google.golang.org/grpc/status"
    "google.golang.org/protobuf/types/known/timestamppb"

    pb "{{ cookiecutter.go_module_path.strip('/') }}/api/go/{{ cookiecutter.project_name.strip() }}/v1"
    "{{ cookiecutter.go_module_path.strip('/') }}/internal/repos/projects"
)

func (s *Service) AddProject(ctx context.Context, req *pb.AddProjectRequest) (*pb.AddProjectResponse, error) {
    now := time.Now()
    
    project, err := s.projectsRepo.AddProject(ctx, projects.AddProjectParams{
        Name:        req.Name,
        Description: req.Description,
        Status:      projects.StatusenumACTIVE,
        CreatedAt:   now,
        UpdatedAt:   now,
    })
    if err != nil {
        return nil, status.Errorf(codes.Internal, "failed to add project: %v", err)
    }

    return &pb.AddProjectResponse{
        Project: convertProject(project),
    }, nil
}

func (s *Service) GetProject(ctx context.Context, req *pb.GetProjectRequest) (*pb.GetProjectResponse, error) {
    project, err := s.projectsRepo.GetProjectById(ctx, req.Id)
    if err != nil {
        return nil, status.Errorf(codes.NotFound, "project not found: %v", err)
    }

    return &pb.GetProjectResponse{
        Project: convertProject(project),
    }, nil
}

func (s *Service) ListProjects(ctx context.Context, req *pb.ListProjectsRequest) (*pb.ListProjectsResponse, error) {
    projectList, err := s.projectsRepo.GetAllProjects(ctx)
    if err != nil {
        return nil, status.Errorf(codes.Internal, "failed to list projects: %v", err)
    }

    pbProjects := make([]*pb.Project, len(projectList))
    for i, p := range projectList {
        pbProjects[i] = convertProject(p)
    }

    return &pb.ListProjectsResponse{
        Projects: pbProjects,
    }, nil
}

// Helper function to convert DB model to protobuf
func convertProject(p projects.Project) *pb.Project {
    return &pb.Project{
        Id:          p.ID,
        Name:        p.Name,
        Description: p.Description,
        Status:      convertStatus(p.Status),
        CreatedAt:   timestamppb.New(p.CreatedAt),
        UpdatedAt:   timestamppb.New(p.UpdatedAt),
    }
}

func convertStatus(s projects.Statusenum) pb.ProjectStatus {
    switch s {
    case projects.StatusenumACTIVE:
        return pb.ProjectStatus_ACTIVE
    case projects.StatusenumDELETED:
        return pb.ProjectStatus_DELETED
    default:
        return pb.ProjectStatus_ACTIVE
    }
}

package svc

import (
    "context"
    "testing"
    "time"

    "github.com/stretchr/testify/assert"
    "google.golang.org/protobuf/types/known/emptypb"

    pb "{{ cookiecutter.go_module_path.strip('/') }}/api/go/{{ cookiecutter.project_name.strip() }}/v1"
)

func (suite *SvcTestSuite) TestAddProject() {
    ctx := context.Background()

    req := &pb.AddProjectRequest{
        Name:        "Test Project",
        Description: "Test Description",
    }

    resp, err := suite.svc.AddProject(ctx, req)
    
    assert.NoError(suite.T(), err)
    assert.NotNil(suite.T(), resp)
    assert.NotNil(suite.T(), resp.Project)
    assert.Equal(suite.T(), "Test Project", resp.Project.Name)
    assert.Greater(suite.T(), resp.Project.Id, int64(0))
}

func (suite *SvcTestSuite) TestListProjects() {
    ctx := context.Background()

    // Add test data
    _, err := suite.svc.AddProject(ctx, &pb.AddProjectRequest{
        Name:        "Project 1",
        Description: "Description 1",
    })
    assert.NoError(suite.T(), err)

    _, err = suite.svc.AddProject(ctx, &pb.AddProjectRequest{
        Name:        "Project 2",
        Description: "Description 2",
    })
    assert.NoError(suite.T(), err)

    // List projects
    resp, err := suite.svc.ListProjects(ctx, &pb.ListProjectsRequest{})
    
    assert.NoError(suite.T(), err)
    assert.NotNil(suite.T(), resp)
    assert.GreaterOrEqual(suite.T(), len(resp.Projects), 2)
}

## 🆘 Troubleshooting

### Build Fails
```bash
# Clean and rebuild
make clean
go mod tidy
make build
```

### Tests Fail
```bash
# Ensure Docker is running (testcontainers requires it)
docker info >/dev/null
make test
```

### Proto Generation Fails
```bash
# Reinstall tools
scripts/install-buf-deps.sh
make proto
```

### Database Issues
```bash
# Restart PostgreSQL
make docker-down
make docker-up
```
